cmake_minimum_required(VERSION 3.20)
project(NoctraLink VERSION 1.0.1 LANGUAGES C)
include(CTest)
if(BUILD_TESTING)
  add_executable(account_id_test tests/account_id_test.c)
  add_test(NAME account_id_conversion COMMAND account_id_test)
endif()
if(WIN32)
  enable_language(RC)
  add_executable(NoctraLink WIN32 src/main.c src/app.rc)
  target_include_directories(NoctraLink PRIVATE assets src)
  target_link_libraries(NoctraLink PRIVATE user32 gdi32 shell32 dwmapi)
  set_property(TARGET NoctraLink PROPERTY C_STANDARD 11)
  if(MSVC)
    target_compile_options(NoctraLink PRIVATE /W4 /utf-8)
    # app.rc already embeds app.manifest as resource 1. Avoid a second manifest.
    target_link_options(NoctraLink PRIVATE /MANIFEST:NO)
    set_property(TARGET NoctraLink PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(NoctraLink PRIVATE -Wall -Wextra)
    target_link_options(NoctraLink PRIVATE -municode)
  endif()
endif()
